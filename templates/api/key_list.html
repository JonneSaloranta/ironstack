# Write

    BASE + "foods/",
    headers=headers,
    json={
        "name": "Chicken breast",
        "serving_size": "100",
        "serving_unit": "g",
        "calories": 165,
        "protein_grams": "31",
        "carbohydrate_grams": "0",
        "fat_grams": "3.6",
    },
)
